\section{算法与定义}
\begin{definition}[迭代停止条件]
当评价指标达到预设阈值或达到最大迭代次数时，结束原型迭代。
\end{definition}
\begin{algorithm}[htbp]
  \caption{原型迭代流程示例}\label{alg:iteration}
  \begin{algorithmic}[1]
    \Require 初始原型 $P_0$，最大迭代次数 $K$
    \Ensure 迭代原型 $P$
    \State $P \gets P_0$
    \For{$k=1,\ldots,K$}
      \State 收集用户评价并分析问题
      \If{达到预设评价标准}
        \State \textbf{break}
      \EndIf
      \State 修改原型 $P$
    \EndFor
    \State \Return $P$
  \end{algorithmic}
\end{algorithm}
