\NewDocumentCommand\imdtsetup{+m}{\keys_set:nn{imdt}{#1}}
\NewDocumentCommand\imdtshowhints{}{\IMDTDrafttrue}
\NewDocumentCommand\imdthidehints{}{\IMDTDraftfalse}
\NewDocumentCommand\imdtshowexamples{}{\IMDTShowExamplestrue}
\NewDocumentCommand\imdthideexamples{}{\IMDTShowExamplesfalse}
\NewDocumentCommand\imdtsubmission{}{
  \imdtsetup{mode=writing,hints=hide,examples=hide,appendix=hide}
}
\NewDocumentCommand\IMDTIncludeChapters{}{
  \clist_map_inline:Nn\g_imdt_chapters_clist{\include{##1}}
}
\NewDocumentCommand\IMDTValidateMetadata{}{
  \int_compare:nNnT{\tl_count:V\ThesisTitle}>{25}{
    \PackageWarning{imdt}{Title~exceeds~25~tokens;~check~config/user.tex}
  }
  \ifIMDTDraft\else
    \clist_map_inline:nn {ThesisTitle,StudentName,StudentID}{
      \exp_args:Nc \tl_if_blank:VTF {##1}{
        \PackageWarning{imdt}{Missing~metadata~##1;~fill~config/user.tex}
      }{}
    }
  \fi
}
