\newcommand\IMDTWordRun[7]{%
  \def\IMDTShouldPaint{1}%
  \ifIMDTHideWordHints
    \ifstrequal{#5}{0,0,0}{}{\def\IMDTShouldPaint{0}}%
  \fi
  \ifnum\IMDTShouldPaint=1\relax
  \put(#1,#2){\hbox{\XeTeXinterchartokenstate=0%
    \fontsize{#4bp}{#4bp}\selectfont\csname IMDTword#3\endcsname
    \ifdim#6bp>0bp\addfontfeature{FakeBold=#6}\fi
    \color[rgb]{#5}#7}}%
  \fi}
\newcommand\IMDTWordRect[5]{%
  \put(#1,#2){\begingroup\color[rgb]{#5}%
    \begin{pgfpicture}\pgfsetbaseline{0pt}%
      \pgfpathrectangle{\pgfpointorigin}{\pgfpoint{#3bp}{#4bp}}%
      \pgfusepath{fill}%
    \end{pgfpicture}\endgroup}}

\newsavebox\IMDTValueBox
\newcommand\IMDTWordValue[5]{%
  \put(#1,#2){\begingroup\songti\fontsize{#4bp}{#4bp}\selectfont
    \sbox\IMDTValueBox{#5}%
    \ifdim\wd\IMDTValueBox>#3bp
      \PackageWarning{imdt}{A fixed form value exceeds its reserved width}%
    \fi
    \makebox[#3bp][c]{\usebox\IMDTValueBox}\endgroup}}

    \fontsize{#4bp}{#4bp}\selectfont\IMDTwordhei\addfontfeature{FakeBold=2.857}%
    \sbox\IMDTValueBox{#5}%
    \ifdim\wd\IMDTValueBox>#3bp
      \PackageWarning{imdt}{A fixed form value exceeds its reserved width}%
    \fi
    \makebox[#3bp][c]{\usebox\IMDTValueBox}\endgroup}}
\newsavebox\IMDTAreaBox
\newcommand\IMDTWordArea[5]{%
  \begingroup\songti\fontsize{10.56bp}{13.68bp}\selectfont
  \setbox\IMDTAreaBox=\vbox{\hsize=#3bp\parindent=0pt\parskip=0pt #5\par}%
  \ifdim\dimexpr\ht\IMDTAreaBox+\dp\IMDTAreaBox\relax>#4bp
    \PackageWarning{imdt}{Text exceeds a fixed form area; shorten it or attach a continuation}%
  \fi
  \put(#1,#2){\raisebox{-\height}{\usebox\IMDTAreaBox}}%
  \endgroup}
